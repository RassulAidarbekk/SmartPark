cmake_minimum_required(VERSION 3.14)
project(SmartPark CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Use a system GLFW if present, otherwise download and build it automatically.
find_package(glfw3 3.3 QUIET)
if(NOT glfw3_FOUND)
  include(FetchContent)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4)
  FetchContent_MakeAvailable(glfw)
endif()

find_package(OpenGL REQUIRED)

find_package(Threads REQUIRED)

# audio.cpp holds the miniaudio implementation (header-only, shipped in this folder)
add_executable(smartpark main.cpp audio.cpp)
target_link_libraries(smartpark PRIVATE glfw OpenGL::GL Threads::Threads)
if(UNIX AND NOT APPLE)
  target_link_libraries(smartpark PRIVATE ${CMAKE_DL_LIBS} m)
endif()
if(APPLE)
  target_link_libraries(smartpark PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
endif()
if(MSVC)
  target_compile_options(smartpark PRIVATE /utf-8)
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT smartpark)
endif()
# Run from the build folder so levels/ and progress file are created next to the exe
set_target_properties(smartpark PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:smartpark>")

include(CTest)
if(BUILD_TESTING)
  add_executable(smartpark_safety_tests test_safety.cpp)
  add_test(NAME layout_safety COMMAND smartpark_safety_tests)
  # drives the whole AUTODROM EXAM headlessly with the real physics + rules (no window needed)
  add_executable(smartpark_exam_test tests/exam_autopilot.cpp audio.cpp)
  target_include_directories(smartpark_exam_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(smartpark_exam_test PRIVATE glfw OpenGL::GL Threads::Threads)
  if(UNIX AND NOT APPLE)
    target_link_libraries(smartpark_exam_test PRIVATE ${CMAKE_DL_LIBS} m)
  endif()
  if(APPLE)
    target_link_libraries(smartpark_exam_test PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
  endif()
  if(MSVC)
    target_compile_options(smartpark_exam_test PRIVATE /utf-8)
  endif()
  add_test(NAME exam_autopilot COMMAND smartpark_exam_test)
endif()
